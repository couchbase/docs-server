
// tag::create-bucket[]
[source,bash]
----
curl -v -X POST http://127.0.0.1:8091/pools/default/buckets \
-u Administrator:password \
-d name=testBucket \
-d ramQuota=100 \
-d encryptionAtRestKeyId=0 
----
// end::create-bucket[]

// tag::alter-bucket[]
[source,bash]
----
curl -v -X POST http://127.0.0.1:8091/pools/default/buckets/testBucket \
-u Administrator:password \
-d encryptionAtRestKeyId=18 
----
// end::alter-bucket[]
